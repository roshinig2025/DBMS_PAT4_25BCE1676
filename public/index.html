// Fetch all complaints from the Vercel/PostgreSQL API async function fetchComplaints() { try { const res = await fetch('/api/complaints'); const data = await res.json(); // Update Stat Cards document.getElementById('statTotal').innerText = data.length; document.getElementById('statPending').innerText = data.filter(c => c.status === 'Pending').length; document.getElementById('statInProgress').innerText = data.filter(c => c.status === 'In Progress').length; document.getElementById('statResolved').innerText = data.filter(c => c.status === 'Resolved').length; // Render Table Rows const tbody = document.getElementById('complaintsTbody'); tbody.innerHTML = ''; data.forEach(c => { const tr = document.createElement('tr'); tr.className = 'hover:bg-slate-800/30 transition'; tr.innerHTML = ` ${c.complaint_id} ${c.title} ${c.category} ${c.priority} ${c.status} ${currentUser && currentUser.role === 'admin' ? ` ` : 'View Only'} `; tbody.appendChild(tr); }); } catch (err) { console.error("Error fetching complaints:", err); } } // Create a new complaint via POST /api/complaints async function handleCreate(e) { e.preventDefault(); await fetch('/api/complaints', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title: document.getElementById('mTitle').value, category: document.getElementById('mCategory').value, priority: document.getElementById('mPriority').value, description: document.getElementById('mDesc').value, userEmail: currentUser.email, userName: currentUser.name }) }); closeModal(); fetchComplaints(); } // Update status via PUT /api/complaints/:id async function updateStatus(id, status) { await fetch(`/api/complaints/${id}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ status }) }); fetchComplaints(); } // Delete complaint via DELETE /api/complaints/:id async function deleteComplaint(id) { await fetch(`/api/complaints/${id}`, { method: 'DELETE' }); fetchComplaints(); }
